235def group_transfers_between_cards(df: pd.DataFrame) -> pd.DataFrame: 

236 """ 

237 groups duplicates of transactions caused by inter-card transfers. 

238 use in get_top_transactions only if not sure what this function does. 

239 :param df: input dataframe with duplicates 

240 :return: dataframe with grouped transactions (only positive payments left) 

241 """ 

242 logger.debug(f"Grouping transfers between cards. Input shape: {df.shape}") 

243 

244 # группируем по описанию, MCC и номеру карты 

245 # если MCC и карта пустые у обеих операций, значит это, возможно, перевод между своими счетами 

246 to_drop = set() 

247 for descr, group in df.groupby("Описание"): 

248 if group["MCC"].isna().all() and group["Номер карты"].isna().all() and len(group) > 1: 

249 # проверяем каждую пару по времени 

250 times = group["Дата операции"].sort_values() 

251 for i, t1 in enumerate(times): 

252 near = group[ 

253 (group["Дата операции"] - t1).dt.total_seconds().abs() <= 2 

254 ] 

255 if len(near) == 2: 

256 # две операции — оставляем положительную 

257 idx_pos = near.loc[near["Сумма платежа"] > 0].index 

258 idx_neg = near.loc[near["Сумма платежа"] < 0].index 

259 if len(idx_pos) == 1 and len(idx_neg) == 1: 

260 to_drop.update(idx_neg.tolist()) 

261 

262 # убираем дубликаты-переводы 

263 if to_drop: 

264 logger.debug(f"Found {len(to_drop)} transfer duplicates to remove") 

265 df = df.drop(index=list(to_drop)) 

266 else: 

267 logger.debug("No transfer duplicates found") 

268 

269 logger.debug(f"After grouping transfers - shape: {df.shape}") 

270 return df

273def get_top_transactions(df: pd.DataFrame, top_n: int = 5) -> pd.DataFrame: 

274 """ 

275 Returns the top N transactions by column "Сумма платежа", 

276 considering only successful ("OK") operations. 

277 Removes duplicates that match the description, time (≤ 2 sec), 

278 and with blank fields "MCC" and "Номер карыт" — leaving the option with a positive amount. 

279 requires group_transfers_between_cards func from src.utils 

280 :param df: input dataframe 

281 :param top_n: the amount of transactions to return 

282 :return: dataframe with top N transactions 

283 """ 

284 logger.info(f"Getting top {top_n} transactions from dataframe. Input shape: {df.shape}") 

285 

286 if len(df) == 0: 

287 logger.warning("Empty dataframe provided for top transactions") 

288 return df 

289 

290 # фильтруем только успешные операции 

291 initial_count = len(df) 

292 df = df[df["Статус"] == "OK"].copy() 

293 successful_count = len(df) 

294 logger.debug(f"Filtered successful transactions: {successful_count}/{initial_count}") 

295 

296 if len(df) == 0: 

297 logger.warning("No successful transactions found") 

298 return df 

299 

300 # выборка из top_n*2 операций. 

301 abs_sum = df["Сумма платежа"].abs().to_numpy() 

302 if len(df) <= top_n * 2: 

303 top_df = df.copy() 

304 logger.debug(f"Using all {len(df)} transactions for analysis") 

305 else: 

306 top_idx = np.argpartition(abs_sum, -top_n * 2)[-top_n * 2:] 

307 top_df = df.iloc[top_idx].copy() 

308 logger.debug(f"Selected top {top_n * 2} transactions for analysis") 

309 

310 top_df = group_transfers_between_cards(top_df) 

311 

312 # сортировка по модулю суммы и дате (подготовка к возврату) 

313 top_df["abs_sum"] = top_df["Сумма платежа"].abs() 

314 top_df = top_df.sort_values( 

315 by=["abs_sum", "Дата операции"], 

316 ascending=[False, False], 

317 kind="mergesort" 

318 ).drop(columns=["abs_sum"]) 

319 

320 # возврат ровно top_n строк 

321 result = top_df.head(top_n).reset_index(drop=True) 

322 logger.info(f"Top {len(result)} transactions prepared successfully") 

323 

324 return result

365 """ 

366 Returns the current rates of the specified currencies relative to base_currency. 

367 The result is a list of format dictionaries: 

368 [ 

369 {"currency": "USD", "rate": 100}, 

370 {"currency": "EUR", "rate": 200} 

371 ] 

372 :param base_currency: the currency to get the rate of 

373 :param symbols: the currencies to get the rate of 

374 :return: a list of format dictionaries: 

375 """ 

376 logger.info(f"Fetching currency rates for {base_currency} -> {symbols}") 

377 

378 if not symbols: 

379 logger.warning("Empty symbols list provided for currency rates") 

380 return [] 

381 

382 params = { 

383 "source": base_currency, 

384 "currencies": ",".join(symbols) 

385 } 

386 

387 data = fetch_from_apilayer("live", params) 

388 quotes = data.get("quotes") 

389 if not quotes: 

390 logger.error("API response does not contain 'quotes' field") 

391 raise RuntimeError("Ответ не содержит поля 'quotes'")

402

404 logger.debug(f"Rates: {rates}")
